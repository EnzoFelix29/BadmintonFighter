    47→    47→function todayISO(): string {
    48→    48→  const d = new Date();
    49→    49→  const y = d.getFullYear();
    50→    50→  const m = String(d.getMonth() + 1).padStart(2, '0');
    51→    51→  const day = String(d.getDate()).padStart(2, '0');
    52→    52→  return `${y}-${m}-${day}`;
    53→    53→}

    56→    56→function formatThaiDate(iso?: string): string {
    57→    57→  if (!iso) return '-';
    58→    58→  const parts = iso.split('-');
    59→    59→  if (parts.length !== 3) return iso;
    60→    60→  const [y, m, d] = parts.map(Number);
    61→    61→  if (!y || !m || !d) return iso;
    62→    62→  return `${d}/${m}/${y + 543}`;
    63→    63→}

   265→   265→  function playTone(
   266→   266→    ctx: AudioContext,
   267→   267→    freq: number,
   268→   268→    start: number,
   269→   269→    duration: number,
   270→   270→    type: OscillatorType = 'sine',
   271→   271→    volume = 0.12
   272→   272→  ) {
   273→   273→    const osc = ctx.createOscillator();
   274→   274→    const gain = ctx.createGain();
   275→   275→    const t0 = ctx.currentTime + start;
   276→   276→    osc.type = type;
   277→   277→    osc.frequency.setValueAtTime(freq, t0);
   278→   278→    gain.gain.setValueAtTime(0.0001, t0);
   279→   279→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   280→   280→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   281→   281→    osc.connect(gain);
   282→   282→    gain.connect(ctx.destination);
   283→   283→    osc.start(t0);
   284→   284→    osc.stop(t0 + duration + 0.05);
   285→   285→  }
