    41→    41→function todayISO(): string {
    42→    42→  const d = new Date();
    43→    43→  const y = d.getFullYear();
    44→    44→  const m = String(d.getMonth() + 1).padStart(2, '0');
    45→    45→  const day = String(d.getDate()).padStart(2, '0');
    46→    46→  return `${y}-${m}-${day}`;
    47→    47→}

    50→    50→function formatThaiDate(iso?: string): string {
    51→    51→  if (!iso) return '-';
    52→    52→  const parts = iso.split('-');
    53→    53→  if (parts.length !== 3) return iso;
    54→    54→  const [y, m, d] = parts.map(Number);
    55→    55→  if (!y || !m || !d) return iso;
    56→    56→  return `${d}/${m}/${y + 543}`;
    57→    57→}

   218→   218→  function playTone(
   219→   219→    ctx: AudioContext,
   220→   220→    freq: number,
   221→   221→    start: number,
   222→   222→    duration: number,
   223→   223→    type: OscillatorType = 'sine',
   224→   224→    volume = 0.12
   225→   225→  ) {
   226→   226→    const osc = ctx.createOscillator();
   227→   227→    const gain = ctx.createGain();
   228→   228→    const t0 = ctx.currentTime + start;
   229→   229→    osc.type = type;
   230→   230→    osc.frequency.setValueAtTime(freq, t0);
   231→   231→    gain.gain.setValueAtTime(0.0001, t0);
   232→   232→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   233→   233→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   234→   234→    osc.connect(gain);
   235→   235→    gain.connect(ctx.destination);
   236→   236→    osc.start(t0);
   237→   237→    osc.stop(t0 + duration + 0.05);
   238→   238→  }
