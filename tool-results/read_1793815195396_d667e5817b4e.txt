    29→// วันที่วันนี้ตามเวลาท้องถิ่น (รูปแบบ yyyy-mm-dd สำหรับ date input)
    30→function todayISO(): string {
    31→  const d = new Date();
    32→  const y = d.getFullYear();
    33→  const m = String(d.getMonth() + 1).padStart(2, '0');
    34→  const day = String(d.getDate()).padStart(2, '0');
    35→  return `${y}-${m}-${day}`;
    36→}

    38→// แปลงเป็นรูปแบบไทย ด/ด/ดดด พ.ศ.
    39→function formatThaiDate(iso?: string): string {
    40→  if (!iso) return '-';
    41→  const parts = iso.split('-');
    42→  if (parts.length !== 3) return iso;
    43→  const [y, m, d] = parts.map(Number);
    44→  if (!y || !m || !d) return iso;
    45→  return `${d}/${m}/${y + 543}`;
    46→}

    49→function rankBadgeClass(index: number): string {

    51→  if (index === 1) return 'bg-gradient-to-br from-slate-300 to-slate-400 text-white shadow-slate-400/40';
    52→  if (index === 2) return 'bg-gradient-to-br from-orange-300 to-amber-700 text-white shadow-orange-400/40';
    53→  return 'bg-gray-100 text-gray-500 dark:bg-gray-800 dark:text-gray-400';
    54→}

   113→  function getAudioCtx(): AudioContext | null {
   114→    try {

   131→  function playTone(
   132→    ctx: AudioContext,
   133→    freq: number,
   134→    start: number,
   135→    duration: number,
   136→    type: OscillatorType = 'sine',
   137→    volume = 0.12
   138→  ) {
   139→    const osc = ctx.createOscillator();
   140→    const gain = ctx.createGain();
   141→    const t0 = ctx.currentTime + start;
   142→    osc.type = type;
   143→    osc.frequency.setValueAtTime(freq, t0);
   144→    gain.gain.setValueAtTime(0.0001, t0);
   145→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   146→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   147→    osc.connect(gain);
   148→    gain.connect(ctx.destination);
   149→    osc.start(t0);
   150→    osc.stop(t0 + duration + 0.05);
   151→  }

   153→  function playSound(kind: SoundKind) {
   154→    if (soundMutedRef.current) return;
   155→    const ctx = getAudioCtx();

   157→    switch (kind) {
   158→      case 'success': // บันทึกสำเร็จ: ไล่โน้ตขึ้น G5 → C6 → E6
   159→        playTone(ctx, 783.99, 0, 0.14);
   160→        playTone(ctx, 1046.5, 0.1, 0.16);
   161→        playTone(ctx, 1318.51, 0.2, 0.3, 'sine', 0.1);
   162→        break;
   163→      case 'error': // กรอกไม่ครบ/ผิดพลาด: เสียงต่ำสั้น 2 จังหวะ
   164→        playTone(ctx, 233.08, 0, 0.15, 'square', 0.05);
   165→        playTone(ctx, 174.61, 0.13, 0.22, 'square', 0.05);
   166→        break;
   167→      case 'open': // เปิดสรุปภาพรวม: โน้ตคู่สั้นเบา
   168→        playTone(ctx, 523.25, 0, 0.09, 'sine', 0.07);
   169→        playTone(ctx, 659.25, 0.07, 0.12, 'sine', 0.07);
   170→        break;
   171→      case 'clear': // ล้างข้อมูล: ไล่โน้ตลง
   172→        playTone(ctx, 659.25, 0, 0.1, 'triangle', 0.07);
   173→        playTone(ctx, 523.25, 0.09, 0.1, 'triangle', 0.07);
   174→        playTone(ctx, 392, 0.18, 0.2, 'triangle', 0.07);
   175→        break;
   176→    }
   177→  }

   221→    const courtVal = court.trim() || '-';

   229→    if (isNaN(sA) || isNaN(sB) || sA < 0 || sB < 0) return showAlert('กรุณากรอกคะแนนให้ถูกต้อง');

   233→      id: matches.length + 1,
   234→      court: courtVal,
   235→      type,
   236→      teamA: teamAVal,
   237→      teamB: teamBVal,
   238→      scoreA: sA,

   240→      maxScore,
   241→      date: matchDate || todayISO(),
   242→    };
   243→
   244→    const newMatches = [...matches, matchData];
   245→    setMatches(newMatches);
   246→
   247→    const newStats = { ...teamStats };
   248→    if (!newStats[teamAVal]) newStats[teamAVal] = { name: teamAVal, played: 0, win: 0, lose: 0 };
   249→    if (!newStats[teamBVal]) newStats[teamBVal] = { name: teamBVal, played: 0, win: 0, lose: 0 };
   250→    newStats[teamAVal] = { ...newStats[teamAVal], played: newStats[teamAVal].played + 1 };
   251→    newStats[teamBVal] = { ...newStats[teamBVal], played: newStats[teamBVal].played + 1 };
   252→    if (sA > sB) {
   253→      newStats[teamAVal] = { ...newStats[teamAVal], win: newStats[teamAVal].win + 1 };
   254→      newStats[teamBVal] = { ...newStats[teamBVal], lose: newStats[teamBVal].lose + 1 };
   255→    } else {
   256→      newStats[teamBVal] = { ...newStats[teamBVal], win: newStats[teamBVal].win + 1 };
   257→      newStats[teamAVal] = { ...newStats[teamAVal], lose: newStats[teamAVal].lose + 1 };
   258→    }
   259→    setTeamStats(newStats);
   260→
   261→    // บันทึกลงเบราว์เซอร์
   262→    saveDataToLocal(newMatches, newStats);

   267→
   268→    showAlert('บันทึกผลสำเร็จ!', false);
   269→  }
