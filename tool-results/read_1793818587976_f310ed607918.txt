    46→// วันที่วันนี้ตามเวลาท้องถิ่น (รูปแบบ yyyy-mm-dd สำหรับ date input)
    47→function todayISO(): string {
    48→  const d = new Date();
    49→  const y = d.getFullYear();
    50→  const m = String(d.getMonth() + 1).padStart(2, '0');
    51→  const day = String(d.getDate()).padStart(2, '0');
    52→  return `${y}-${m}-${day}`;
    53→}

    55→// แปลงเป็นรูปแบบไทย ด/ด/ดดด พ.ศ.
    56→function formatThaiDate(iso?: string): string {
    57→  if (!iso) return '-';
    58→  const parts = iso.split('-');
    59→  if (parts.length !== 3) return iso;
    60→  const [y, m, d] = parts.map(Number);
    61→  if (!y || !m || !d) return iso;
    62→  return `${d}/${m}/${y + 543}`;
    63→}

   265→  function playTone(
   266→    ctx: AudioContext,
   267→    freq: number,
   268→    start: number,
   269→    duration: number,
   270→    type: OscillatorType = 'sine',
   271→    volume = 0.12
   272→  ) {
   273→    const osc = ctx.createOscillator();
   274→    const gain = ctx.createGain();
   275→    const t0 = ctx.currentTime + start;
   276→    osc.type = type;
   277→    osc.frequency.setValueAtTime(freq, t0);
   278→    gain.gain.setValueAtTime(0.0001, t0);
   279→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   280→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   281→    osc.connect(gain);
   282→    gain.connect(ctx.destination);
   283→    osc.start(t0);
   284→    osc.stop(t0 + duration + 0.05);
   285→  }

   287→  function playSound(kind: SoundKind) {
   288→    if (soundMutedRef.current) return;
   289→    const ctx = getAudioCtx();
   290→    if (!ctx) return;
   291→    switch (kind) {
   292→      case 'success': // บันทึกสำเร็จ: เสียงเหรียญเกม 8-bit ไล่โน้ตขึ้น E5 → G5 → C6
   293→        playTone(ctx, 659.25, 0, 0.08, 'square', 0.05);
   294→        playTone(ctx, 783.99, 0.08, 0.08, 'square', 0.05);
   295→        playTone(ctx, 1046.5, 0.16, 0.25, 'square', 0.05);
   296→        break;
   297→      case 'select': // เลือกตัวละคร: บลิปสั้นแบบ 8-bit
   298→        playTone(ctx, 880, 0, 0.06, 'square', 0.045);
   299→        playTone(ctx, 1174.66, 0.06, 0.1, 'square', 0.045);
   300→        break;
   301→      case 'error': // กรอกไม่ครบ/ผิดพลาด: เสียงต่ำสั้น 2 จังหวะ
   302→        playTone(ctx, 233.08, 0, 0.15, 'square', 0.05);
   303→        playTone(ctx, 174.61, 0.13, 0.22, 'square', 0.05);
   304→        break;
   305→      case 'open': // เปิดสรุปภาพรวม: โน้ตคู่สั้นเบา
   306→        playTone(ctx, 523.25, 0, 0.09, 'sine', 0.07);
   307→        playTone(ctx, 659.25, 0.07, 0.12, 'sine', 0.07);
   308→        break;
   309→      case 'clear': // ล้างข้อมูล: ไล่โน้ตลง
   310→        playTone(ctx, 659.25, 0, 0.1, 'triangle', 0.07);
   311→        playTone(ctx, 523.25, 0.09, 0.1, 'triangle', 0.07);
   312→        playTone(ctx, 392, 0.18, 0.2, 'triangle', 0.07);
   313→        break;
   314→      case 'tick': // เสียงรีลหมุนทีละช่อง (เบา ๆ)
   315→        playTone(ctx, 1318.5, 0, 0.03, 'square', 0.018);
   316→        break;
   317→    }
   318→  }
