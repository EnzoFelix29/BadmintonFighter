    71→function todayISO(): string {
    72→  const d = new Date();
    73→  const y = d.getFullYear();
    74→  const m = String(d.getMonth() + 1).padStart(2, '0');
    75→  const day = String(d.getDate()).padStart(2, '0');
    76→  return `${y}-${m}-${day}`;
    77→}

    79→// แปลงเป็นรูปแบบไทย ด/ด/ดดด พ.ศ.
    80→function formatThaiDate(iso?: string): string {
    81→  if (!iso) return '-';
    82→  const parts = iso.split('-');
    83→  if (parts.length !== 3) return iso;
    84→  const [y, m, d] = parts.map(Number);
    85→  if (!y || !m || !d) return iso;
    86→  return `${d}/${m}/${y + 543}`;
    87→}

   311→  function playTone(
   312→    ctx: AudioContext,
   313→    freq: number,
   314→    start: number,
   315→    duration: number,
   316→    type: OscillatorType = 'sine',
   317→    volume = 0.12
   318→  ) {
   319→    const osc = ctx.createOscillator();
   320→    const gain = ctx.createGain();
   321→    const t0 = ctx.currentTime + start;
   322→    osc.type = type;
   323→    osc.frequency.setValueAtTime(freq, t0);
   324→    gain.gain.setValueAtTime(0.0001, t0);
   325→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   326→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   327→    osc.connect(gain);
   328→    gain.connect(ctx.destination);
   329→    osc.start(t0);
   330→    osc.stop(t0 + duration + 0.05);
   331→  }

   333→  function playSound(kind: SoundKind) {
   334→    if (soundMutedRef.current) return;
   335→    const ctx = getAudioCtx();
   336→    if (!ctx) return;
   337→    switch (kind) {
   338→      case 'success': // บันทึกสำเร็จ: เสียงเหรียญเกม 8-bit ไล่โน้ตขึ้น E5 → G5 → C6
   339→        playTone(ctx, 659.25, 0, 0.08, 'square', 0.05);
   340→        playTone(ctx, 783.99, 0.08, 0.08, 'square', 0.05);
   341→        playTone(ctx, 1046.5, 0.16, 0.25, 'square', 0.05);
   342→        break;
   343→      case 'select': // เลือกตัวละคร: บลิปสั้นแบบ 8-bit
   344→        playTone(ctx, 880, 0, 0.06, 'square', 0.045);
   345→        playTone(ctx, 1174.66, 0.06, 0.1, 'square', 0.045);
   346→        break;
   347→      case 'error': // กรอกไม่ครบ/ผิดพลาด: เสียงต่ำสั้น 2 จังหวะ
   348→        playTone(ctx, 233.08, 0, 0.15, 'square', 0.05);
   349→        playTone(ctx, 174.61, 0.13, 0.22, 'square', 0.05);
   350→        break;
   351→      case 'open': // เปิดสรุปภาพรวม: โน้ตคู่สั้นเบา
   352→        playTone(ctx, 523.25, 0, 0.09, 'sine', 0.07);
   353→        playTone(ctx, 659.25, 0.07, 0.12, 'sine', 0.07);
   354→        break;
   355→      case 'clear': // ล้างข้อมูล: ไล่โน้ตลง
   356→        playTone(ctx, 659.25, 0, 0.1, 'triangle', 0.07);
   357→        playTone(ctx, 523.25, 0.09, 0.1, 'triangle', 0.07);
   358→        playTone(ctx, 392, 0.18, 0.2, 'triangle', 0.07);
   359→        break;
   360→      case 'tick': // เสียงรีลหมุนทีละช่อง (เบา ๆ)
   361→        playTone(ctx, 1318.5, 0, 0.03, 'square', 0.018);
   362→        break;
   363→    }
   364→  }

   573→  function saveMatch() {
   574→    const courtVal = court.trim() || '-';
   575→    const isDoubles = type === 'คู่';
   576→    const slots: PlayerSlot[] = isDoubles ? [slotA1, slotA2, slotB1, slotB2] : [slotA1, slotB1];
   577→    const sA = parseInt(scoreA);
   578→    const sB = parseInt(scoreB);
   579→
   580→    if (players.length === 0)
   581→      return showAlert('ยังไม่มีผู้เล่น — กด "＋ ลงทะเบียนผู้เล่นใหม่" ที่ช่องชื่อก่อน');
   582→    // ทุกช่องต้องเลือกผู้เล่นที่ลงทะเบียนไว้
   583→    if (slots.some((s) => !s.name)) return showAlert('กรุณาเลือกผู้เล่นให้ครบทุกช่อง');
   584→    const names = slots.map((s) => s.name);
   585→    if (new Set(names).size !== names.length)
   586→      return showAlert('ผู้เล่นต้องไม่ซ้ำกันในแมตช์เดียวกัน');
   587→    const profs: PlayerProfile[] = [];
   588→    for (const n of names) {
   589→      const p = players.find((x) => x.name === n);
   590→      if (!p) return showAlert('ไม่พบโปรไฟล์ผู้เล่น กรุณาลงทะเบียนใหม่');
   591→      profs.push(p);
   592→    }
   593→    const skinIds = profs.map((p) => p.skinId);
   594→    if (new Set(skinIds).size !== skinIds.length)
   595→      return showAlert('สกินตัวละครแต่ละคนต้องไม่ซ้ำกันในแมตช์เดียวกัน');
   596→    if (isNaN(sA) || isNaN(sB) || sA < 0 || sB < 0) return showAlert('กรุณากรอกคะแนนให้ถูกต้อง');
   597→    if (sA === sB) return showAlert('แบดมินตันไม่มีผลเสมอ กรุณาตรวจสอบคะแนน');
   598→
   599→    const playersA: PlayerInfo[] = isDoubles
   600→      ? [
   601→          { name: profs[0].name, skinId: profs[0].skinId, racketId: profs[0].racketId, shoeId: profs[0].shoeId },
   602→          { name: profs[1].name, skinId: profs[1].skinId, racketId: profs[1].racketId, shoeId: profs[1].shoeId },
   603→        ]
   604→      : [{ name: profs[0].name, skinId: profs[0].skinId, racketId: profs[0].racketId, shoeId: profs[0].shoeId }];
   605→    const playersB: PlayerInfo[] = isDoubles
   606→      ? [
   607→          { name: profs[2].name, skinId: profs[2].skinId, racketId: profs[2].racketId, shoeId: profs[2].shoeId },
   608→          { name: profs[3].name, skinId: profs[3].skinId, racketId: profs[3].racketId, shoeId: profs[3].shoeId },
   609→        ]
   610→      : [{ name: profs[1].name, skinId: profs[1].skinId, racketId: profs[1].racketId, shoeId: profs[1].shoeId }];
   611→
   612→    // ชื่อทีม: เดี่ยว = ชื่อผู้เล่น, คู่ = "ชื่อ1 / ชื่อ2"
   613→    const teamAVal = isDoubles ? `${names[0]} / ${names[1]}` : names[0];
   614→    const teamBVal = isDoubles ? `${names[2]} / ${names[3]}` : names[1];
   615→
   616→    const matchData: Match = {
   617→      id: matches.length + 1,
   618→      court: courtVal,
   619→      type,
   620→      teamA: teamAVal,
   621→      teamB: teamBVal,
   622→      playersA,
   623→      playersB,
   624→      scoreA: sA,
   625→      scoreB: sB,
   626→      maxScore,
   627→      date: matchDate || todayISO(),
   628→    };
   629→
   630→    const newMatches = [...matches, matchData];
   631→    setMatches(newMatches);
   632→
   633→    const newStats = { ...teamStats };
   634→    if (!newStats[teamAVal]) newStats[teamAVal] = { name: teamAVal, played: 0, win: 0, lose: 0 };
   635→    if (!newStats[teamBVal]) newStats[teamBVal] = { name: teamBVal, played: 0, win: 0, lose: 0 };
   636→    newStats[teamAVal] = { ...newStats[teamAVal], played: newStats[teamAVal].played + 1 };
   637→    newStats[teamBVal] = { ...newStats[teamBVal], played: newStats[teamBVal].played + 1 };
   638→    if (sA > sB) {
   639→      newStats[teamAVal] = { ...newStats[teamAVal], win: newStats[teamAVal].win + 1 };
   640→      newStats[teamBVal] = { ...newStats[teamBVal], lose: newStats[teamBVal].lose + 1 };
   641→    } else {
   642→      newStats[teamBVal] = { ...newStats[teamBVal], win: newStats[teamBVal].win + 1 };
   643→      newStats[teamAVal] = { ...newStats[teamAVal], lose: newStats[teamAVal].lose + 1 };
   644→    }
   645→    setTeamStats(newStats);
   646→
   647→    // บันทึกลงเบราว์เซอร์
   648→    saveDataToLocal(newMatches, newStats);
   649→
   650→    setScoreA('');
   651→    setScoreB('');
   652→    setMatchDate(todayISO()); // รีเซ็ตกลับเป็นวันนี้สำหรับแมตช์ถัดไป
   653→
   654→    // เหรียญรายบุคคล: ผู้เล่นทีมชนะ +3 เหรียญ/คน ทีมแพ้ +1 เหรียญ/คน
   655→    const split = isDoubles ? 2 : 1;
   656→    const winnerNames = sA > sB ? names.slice(0, split) : names.slice(split);
   657→    const loserNames = sA > sB ? names.slice(split) : names.slice(0, split);
   658→    setPlayers((ps) => awardCoins(ps, winnerNames, loserNames, 3, 1).players);
   659→    showAlert(
   660→      `บันทึกผลสำเร็จ! ชนะ: ${winnerNames.join(', ')} +3 เหรียญ • แพ้: ${loserNames.join(', ')} +1 เหรียญ`,
   661→      false
   662→    );
