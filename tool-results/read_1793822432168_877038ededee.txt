    69→    69→function formatThaiDate(iso?: string): string {
    70→    70→  if (!iso) return '-';
    71→    71→  const parts = iso.split('-');
    72→    72→  if (parts.length !== 3) return iso;
    73→    73→  const [y, m, d] = parts.map(Number);
    74→    74→  if (!y || !m || !d) return iso;
    75→    75→  return `${d}/${m}/${y + 543}`;
    76→    76→}

   298→   298→  function playTone(
   299→   299→    ctx: AudioContext,
   300→   300→    freq: number,
   301→   301→    start: number,
   302→   302→    duration: number,
   303→   303→    type: OscillatorType = 'sine',
   304→   304→    volume = 0.12
   305→   305→  ) {
   306→   306→    const osc = ctx.createOscillator();
   307→   307→    const gain = ctx.createGain();
   308→   308→    const t0 = ctx.currentTime + start;
   309→   309→    osc.type = type;
   310→   310→    osc.frequency.setValueAtTime(freq, t0);
   311→   311→    gain.gain.setValueAtTime(0.0001, t0);
   312→   312→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   313→   313→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   314→   314→    osc.connect(gain);
   315→   315→    gain.connect(ctx.destination);
   316→   316→    osc.start(t0);
   317→   317→    osc.stop(t0 + duration + 0.05);
   318→   318→  }
