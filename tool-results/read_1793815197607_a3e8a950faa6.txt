    30→    30→function todayISO(): string {
    31→    31→  const d = new Date();
    32→    32→  const y = d.getFullYear();
    33→    33→  const m = String(d.getMonth() + 1).padStart(2, '0');
    34→    34→  const day = String(d.getDate()).padStart(2, '0');
    35→    35→  return `${y}-${m}-${day}`;
    36→    36→}

    38→    38→// แปลงเป็นรูปแบบไทย ด/ด/ดดด พ.ศ.
    39→    39→function formatThaiDate(iso?: string): string {
    40→    40→  if (!iso) return '-';
    41→    41→  const parts = iso.split('-');
    42→    42→  if (parts.length !== 3) return iso;
    43→    43→  const [y, m, d] = parts.map(Number);
    44→    44→  if (!y || !m || !d) return iso;
    45→    45→  return `${d}/${m}/${y + 543}`;
    46→    46→}

   131→   131→  function playTone(
   132→   132→    ctx: AudioContext,
   133→   133→    freq: number,
   134→   134→    start: number,
   135→   135→    duration: number,
   136→   136→    type: OscillatorType = 'sine',
   137→   137→    volume = 0.12
   138→   138→  ) {
   139→   139→    const osc = ctx.createOscillator();
   140→   140→    const gain = ctx.createGain();
   141→   141→    const t0 = ctx.currentTime + start;
   142→   142→    osc.type = type;
   143→   143→    osc.frequency.setValueAtTime(freq, t0);
   144→   144→    gain.gain.setValueAtTime(0.0001, t0);
   145→   145→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   146→   146→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   147→   147→    osc.connect(gain);
   148→   148→    gain.connect(ctx.destination);
   149→   149→    osc.start(t0);
   150→   150→    osc.stop(t0 + duration + 0.05);
   151→   151→  }
