    60→function todayISO(): string {
    61→  const d = new Date();
    62→  const y = d.getFullYear();
    63→  const m = String(d.getMonth() + 1).padStart(2, '0');
    64→  const day = String(d.getDate()).padStart(2, '0');
    65→  return `${y}-${m}-${day}`;
    66→}

    68→// แปลงเป็นรูปแบบไทย ด/ด/ดดด พ.ศ.
    69→function formatThaiDate(iso?: string): string {
    70→  if (!iso) return '-';
    71→  const parts = iso.split('-');
    72→  if (parts.length !== 3) return iso;
    73→  const [y, m, d] = parts.map(Number);
    74→  if (!y || !m || !d) return iso;
    75→  return `${d}/${m}/${y + 543}`;
    76→}

   298→  function playTone(
   299→    ctx: AudioContext,
   300→    freq: number,
   301→    start: number,
   302→    duration: number,
   303→    type: OscillatorType = 'sine',
   304→    volume = 0.12
   305→  ) {
   306→    const osc = ctx.createOscillator();
   307→    const gain = ctx.createGain();
   308→    const t0 = ctx.currentTime + start;
   309→    osc.type = type;
   310→    osc.frequency.setValueAtTime(freq, t0);
   311→    gain.gain.setValueAtTime(0.0001, t0);
   312→    gain.gain.exponentialRampToValueAtTime(volume, t0 + 0.02);
   313→    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   314→    osc.connect(gain);
   315→    gain.connect(ctx.destination);
   316→    osc.start(t0);
   317→    osc.stop(t0 + duration + 0.05);
   318→  }

   320→  function playSound(kind: SoundKind) {
   321→    if (soundMutedRef.current) return;
   322→    const ctx = getAudioCtx();
   323→    if (!ctx) return;
   324→    switch (kind) {
   325→      case 'success': // บันทึกสำเร็จ: เสียงเหรียญเกม 8-bit ไล่โน้ตขึ้น E5 → G5 → C6
   326→        playTone(ctx, 659.25, 0, 0.08, 'square', 0.05);
   327→        playTone(ctx, 783.99, 0.08, 0.08, 'square', 0.05);
   328→        playTone(ctx, 1046.5, 0.16, 0.25, 'square', 0.05);
   329→        break;
   330→      case 'select': // เลือกตัวละคร: บลิปสั้นแบบ 8-bit
   331→        playTone(ctx, 880, 0, 0.06, 'square', 0.045);
   332→        playTone(ctx, 1174.66, 0.06, 0.1, 'square', 0.045);
   333→        break;
   334→      case 'error': // กรอกไม่ครบ/ผิดพลาด: เสียงต่ำสั้น 2 จังหวะ
   335→        playTone(ctx, 233.08, 0, 0.15, 'square', 0.05);
   336→        playTone(ctx, 174.61, 0.13, 0.22, 'square', 0.05);
   337→        break;
   338→      case 'open': // เปิดสรุปภาพรวม: โน้ตคู่สั้นเบา
   339→        playTone(ctx, 523.25, 0, 0.09, 'sine', 0.07);
   340→        playTone(ctx, 659.25, 0.07, 0.12, 'sine', 0.07);
   341→        break;
   342→      case 'clear': // ล้างข้อมูล: ไล่โน้ตลง
   343→        playTone(ctx, 659.25, 0, 0.1, 'triangle', 0.07);
   344→        playTone(ctx, 523.25, 0.09, 0.1, 'triangle', 0.07);
   345→        playTone(ctx, 392, 0.18, 0.2, 'triangle', 0.07);
   346→        break;
   347→      case 'tick': // เสียงรีลหมุนทีละช่อง (เบา ๆ)
   348→        playTone(ctx, 1318.5, 0, 0.03, 'square', 0.018);
   349→        break;
   350→    }
   351→  }

   450→      }

   730→  async function saveMatch() {
   731→    if (!loaded) return showAlert('กำลังโหลดข้อมูลจากเซิร์ฟเวอร์ ลองอีกครั้งในอีกสักครู่');

   735→    const sA = parseInt(scoreA);

   737→
   738→    if (players.length === 0)
   739→      return showAlert('ยังไม่มีผู้เล่น — กด "＋ ลงทะเบียนผู้เล่นใหม่" ที่ช่องชื่อก่อน');
   740→    // ทุกช่องต้องเลือกผู้เล่นที่ลงทะเบียนไว้
   741→    if (slots.some((s) => !s.name)) return showAlert('กรุณาเลือกผู้เล่นให้ครบทุกช่อง');
   742→    const names = slots.map((s) => s.name);
   743→    if (new Set(names).size !== names.length)
   744→      return showAlert('ผู้เล่นต้องไม่ซ้ำกันในแมตช์เดียวกัน');
   745→    const profs: PlayerProfile[] = [];
   746→    for (const n of names) {
   747→      const p = players.find((x) => x.name === n);
   748→      if (!p) return showAlert('ไม่พบโปรไฟล์ผู้เล่น กรุณาลงทะเบียนใหม่');
   749→      profs.push(p);
   750→    }
   751→    const skinIds = profs.map((p) => p.skinId);
   752→    if (new Set(skinIds).size !== skinIds.length)
   753→      return showAlert('สกินตัวละครแต่ละคนต้องไม่ซ้ำกันในแมตช์เดียวกัน');
   754→    if (isNaN(sA) || isNaN(sB) || sA < 0 || sB < 0) return showAlert('กรุณากรอกคะแนนให้ถูกต้อง');
   755→    if (sA === sB) return showAlert('แบดมินตันไม่มีผลเสมอ กรุณาตรวจสอบคะแนน');
   756→
   757→    const playersA: PlayerInfo[] = isDoubles
   758→      ? [
   759→          { name: profs[0].name, skinId: profs[0].skinId, racketId: profs[0].racketId, shoeId: profs[0].shoeId },
   760→          { name: profs[1].name, skinId: profs[1].skinId, racketId: profs[1].racketId, shoeId: profs[1].shoeId },
   761→        ]
   762→      : [{ name: profs[0].name, skinId: profs[0].skinId, racketId: profs[0].racketId, shoeId: profs[0].shoeId }];
   763→    const playersB: PlayerInfo[] = isDoubles
   764→      ? [
   765→          { name: profs[2].name, skinId: profs[2].skinId, racketId: profs[2].racketId, shoeId: profs[2].shoeId },
   766→          { name: profs[3].name, skinId: profs[3].skinId, racketId: profs[3].racketId, shoeId: profs[3].shoeId },
   767→        ]
   768→      : [{ name: profs[1].name, skinId: profs[1].skinId, racketId: profs[1].racketId, shoeId: profs[1].shoeId }];
   769→
   770→    // ชื่อทีม: เดี่ยว = ชื่อผู้เล่น, คู่ = "ชื่อ1 / ชื่อ2"
   771→    const teamAVal = isDoubles ? `${names[0]} / ${names[1]}` : names[0];
   772→    const teamBVal = isDoubles ? `${names[2]} / ${names[3]}` : names[1];
   773→
   774→    const matchData: Match = {
   775→      id: matches.length + 1,
   776→      court: courtVal,
   777→      type,
   778→      teamA: teamAVal,
   779→      teamB: teamBVal,
   780→      playersA,
   781→      playersB,
   782→      scoreA: sA,
   783→      scoreB: sB,
   784→      maxScore,
   785→      date: matchDate || todayISO(),
   786→    };
   787→
   788→    // เหรียญรายบุคคล: ผู้เล่นทีมชนะ +3 เหรียญ/คน ทีมแพ้ +1 เหรียญ/คน (เซิร์ฟเวอร์คำนวณและบันทึกให้)
   789→    const split = isDoubles ? 2 : 1;
   790→    const winnerNames = sA > sB ? names.slice(0, split) : names.slice(split);
   791→    const loserNames = sA > sB ? names.slice(split) : names.slice(0, split);
